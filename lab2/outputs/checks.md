| :-- | --: | --: | --: | --: | --: | --: | :-: |
| +∞ | 0 | 0 | 2 | 4 | 20 | 20 | так |
| 0,9 | 0 | 1 | 2 | 3 | 21 | 21 | так |
| 0,7 | 1 | 2 | 1 | 2 | 12 | 12 | так |
| 0,4 | 2 | 2 | 0 | 2 | 2 | 2 | так |
| 0,2 | 2 | 3 | 0 | 1 | 3 | 3 | так |
| 0,1 | 2 | 4 | 0 | 0 | 4 | 4 | так |
